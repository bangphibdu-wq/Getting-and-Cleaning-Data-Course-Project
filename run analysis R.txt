# Tải các gói thư viện cần thiết để tối ưu hiệu năng và xử lý dữ liệu
library(data.table)
library(dplyr)

# ==============================================================================
# BƯỚC 0: TẢI VÀ GIẢI NÉN DỮ LIỆU CHUẨN
# ==============================================================================
zip_url <- "https://d396qusza40orc.cloudfront.net/getdata%2Fprojectfiles%2FUCI%20HAR%20Dataset.zip"
if (!file.exists("Dataset.zip")) {
  download.file(zip_url, destfile = "Dataset.zip")
}
if (!file.exists("UCI HAR Dataset")) {
  unzip("Dataset.zip")
}

# ==============================================================================
# BƯỚC 1: GỘP TẬP DỮ LIỆU TRAIN VÀ TEST THÀNH MỘT (TỐI ƯU BẰNG FREAD)
# ==============================================================================
# Đọc danh sách tên cột và nhãn hoạt động thô
features <- fread("UCI HAR Dataset/features.txt", col.names = c("n","functions"))
activities <- fread("UCI HAR Dataset/activity_labels.txt", col.names = c("code", "activity"))

# Đọc tập Test - BẮT BUỘC dùng check.names = FALSE để giữ nguyên ký tự thô (-) và ()
subject_test <- fread("UCI HAR Dataset/test/subject_test.txt", col.names = "subject")
x_test <- fread("UCI HAR Dataset/test/X_test.txt", col.names = features$functions, check.names = FALSE)
y_test <- fread("UCI HAR Dataset/test/y_test.txt", col.names = "code")

# Đọc tập Train - BẮT BUỘC dùng check.names = FALSE
subject_train <- fread("UCI HAR Dataset/train/subject_train.txt", col.names = "subject")
x_train <- fread("UCI HAR Dataset/train/X_train.txt", col.names = features$functions, check.names = FALSE)
y_train <- fread("UCI HAR Dataset/train/y_train.txt", col.names = "code")

# Gộp dữ liệu hiệu năng cao
X_total <- rbind(x_train, x_test)
Y_total <- rbind(y_train, y_test)
Subject_total <- rbind(subject_train, subject_test)
Merged_Data <- cbind(Subject_total, Y_total, X_total)

# ==============================================================================
# BƯỚC 2: CHỈ GIỮ LẠI PHÉP ĐO TRUNG BÌNH (MEAN) VÀ ĐỘ LỆCH CHUẨN (STD)
# ==============================================================================
# Lúc này tên cột vẫn giữ nguyên định dạng thô dạng "-mean()" nên lọc contains rất chuẩn
TidyData <- Merged_Data %>% 
  select(subject, code, contains("-mean()"), contains("-std()"))

# ==============================================================================
# BƯỚC 3: ĐẶT TÊN HOẠT ĐỘNG DỄ HIỂU
# ==============================================================================
TidyData$code <- activities[TidyData$code, 2]$activity

# ==============================================================================
# BƯỚC 4: LÀM SẠCH TÊN BIẾN SỐ (Xử lý chuỗi triệt để từ chuỗi gốc)
# ==============================================================================
names(TidyData)[2] = "activity"

# Xử lý phần thân chữ viết tắt trước
names(TidyData) <- gsub("Acc", "Accelerometer", names(TidyData))
names(TidyData) <- gsub("Gyro", "Gyroscope", names(TidyData))
names(TidyData) <- gsub("BodyBody", "Body", names(TidyData))
names(TidyData) <- gsub("Mag", "Magnitude", names(TidyData))
names(TidyData) <- gsub("^t", "Time", names(TidyData))
names(TidyData) <- gsub("^f", "Frequency", names(TidyData))

# Xử lý cấu trúc mean/std gốc (Vì giữ được chữ gốc nên các lệnh dưới đây sẽ hoạt động)
names(TidyData) <- gsub("-mean\\(\\)", "Mean", names(TidyData))
names(TidyData) <- gsub("-std\\(\\)", "Std", names(TidyData))

# Dọn dẹp các ký tự gạch ngang còn sót lại nếu có ở trục tọa độ (ví dụ -X thành X)
names(TidyData) <- gsub("-", "", names(TidyData))

# ==============================================================================
# BƯỚC 5: TẠO BỘ DỮ LIỆU ĐỘC LẬP THỨ HAI VÀ XUẤT FILE TỆP TIN
# ==============================================================================
FinalData <- TidyData %>%
  group_by(subject, activity) %>%
  summarise_all(list(mean = mean))

# Thêm tiền tố vào cột đã tính mean để tường minh thông tin
names(FinalData)[-(1:2)] <- paste0("average_", names(FinalData)[-(1:2)])
# ĐỀ XUẤT MỚI: Đảm bảo các tiền tố mới thêm vào cũng tuân thủ định dạng viết thường
names(FinalData) <- tolower(names(FinalData))
# Xuất dữ liệu sạch
write.table(FinalData, "FinalTidyData.txt", row.name = FALSE)
names(FinalData) <- tolower(names(FinalData))

print("--- HOÀN THÀNH XỬ LÝ TỐI ƯU! FILE FinalTidyData.txt ĐÃ ĐƯỢC TẠO ---")
